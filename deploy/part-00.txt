function currentPath(): array {
    $path = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH) ?: '/';
    $parts = array_values(array_filter(explode('/', trim($path,'/')), fn($v)=>$v!==''));
    $lang = (($parts[0] ?? '') === 'vi') ? 'vi' : 'en';
    if ($lang==='vi') array_shift($parts);
    return [$lang, $parts];
}
function jsonStorePath(): string { return dirname(__DIR__) . '/private/mankind-appointments.json'; }
function saveAppointment(array $row): bool {
    $file=jsonStorePath(); $dir=dirname($file);
    if (!is_dir($dir)) @mkdir($dir,0750,true);
    $fh=@fopen($file,'c+'); if(!$fh) return false;
    $ok=false;
    if(flock($fh,LOCK_EX)){
        $raw=stream_get_contents($fh); $rows=$raw?json_decode($raw,true):[]; if(!is_array($rows))$rows=[];
        $rows[]=$row; rewind($fh); ftruncate($fh,0); $ok=fwrite($fh,json_encode($rows,JSON_PRETTY_PRINT|JSON_UNESCAPED_SLASHES|JSON_UNESCAPED_UNICODE))!==false; fflush($fh); flock($fh,LOCK_UN);
    }
    fclose($fh); return $ok;
}
